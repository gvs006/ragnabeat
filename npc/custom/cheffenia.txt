//===== O PASSE ==============================================

//= Quem expulsa e o Relogio de Cheffenia, que varre o mapa a
//= cada 30 segundos. Varredura em vez de addtimer porque

//= o passe vencido nunca expiraria. A varredura nao esquece.

//======================================================
// O Guarda - a porta, em Prontera
//======================================================
prontera,169,186,4	script	Guarda de Cheffenia::CheffeniaPorta	723,{
	.@resta = cheffenia_ate - gettimetick(2);

	mes "[Guarda de Cheffenia]";

	// Passe ainda de pe: entra de novo sem pagar. Ver O PASSE
	// no cabecalho - o custo e da sessao, nao da porta.
	if (.@resta > 0) {
		mes "Seu passe ainda vale ^0000FF" + (.@resta / 60 + 1) + " minuto(s)^000000.";
		mes "Entre à vontade.";
		next;
		if (select("Voltar para Cheffenia:Fico por aqui") == 2)
			close;
		if (getmapusers(.mapa$) >= .lotacao) {
			mes "[Guarda de Cheffenia]";
			mes "O salão está lotado. Espere alguém sair.";
			close;
		}
		close2;
		warp .mapa$, .x, .y;
		end;
	}

	mes "Atrás de mim fica ^FF0000Cheffenia^000000, o salão onde";
	mes "os senhores de Midgard estão presos.";
	mes "Só há MVP lá dentro. Mais nada.";
	next;
	mes "[Guarda de Cheffenia]";
	mes "A passagem custa ^FF0000" + callfunc("F_InsertComma", .custo) + " zeny^000000";
	mes "e vale por ^0000FF" + .minutos + " minutos^000000.";
	mes " ";
	mes "Se você morrer lá dentro, o passe continua";
	mes "valendo. Volte aqui que eu deixo entrar.";
	next;

	switch (select("Quero entrar:Quantos estão lá dentro?:Melhor não")) {
	case 1:
		mes "[Guarda de Cheffenia]";
		if (BaseLevel < .nivel) {
			mes "Nível base ^FF0000" + .nivel + "^000000, no mínimo.";
			mes "Você tem " + BaseLevel + ". Volte mais forte.";
			close;
		}
		if (getmapusers(.mapa$) >= .lotacao) {
			mes "O salão está lotado - ^FF0000" + .lotacao + "^000000 lá dentro.";
			mes "Espere alguém sair.";
			close;
		}
		if (Zeny < .custo) {
			mes "Faltam ^FF0000" + callfunc("F_InsertComma", .custo - Zeny) + " zeny^000000.";
			close;
		}
		Zeny -= .custo;
		cheffenia_ate = gettimetick(2) + .minutos * 60;
		mes "Que Odin tenha piedade.";
		close2;
		warp .mapa$, .x, .y;
		end;

	case 2:
		mes "[Guarda de Cheffenia]";
		mes "Há ^0000FF" + getmapusers(.mapa$) + "^000000 de " + .lotacao + " lá dentro agora.";
		close;

	case 3:
		mes "[Guarda de Cheffenia]";
		mes "Sensato.";
		close;
	}

OnInit:
	// ---- a configuracao inteira mora aqui ----
	// O Vigia e o Relogio leem daqui por getvariableofnpc, entao
	// mudar um valor neste bloco muda o sistema todo.
	.mapa$ = "ordeal_a02";
	.x = 103;
	.y = 156;
	.custo = 1000000;
	.minutos = 60;
	.nivel = 80;
	.lotacao = 30;
	.volta$ = "prontera";
	.volta_x = 155;
	.volta_y = 186;

	waitingroom "Cheffenia", 0;
	end;
}

//======================================================
// O Vigia - a saida, dentro do salao
//======================================================
ordeal_a02,106,156,4	script	Vigia de Cheffenia::CheffeniaVigia	778,{
	.@resta = cheffenia_ate - gettimetick(2);

	mes "[Vigia de Cheffenia]";
	if (.@resta > 0)
		mes "Restam ^0000FF" + (.@resta / 60 + 1) + " minuto(s)^000000 do seu passe.";
	else
		mes "Seu passe venceu. Sai andando ou eu te arrasto.";
	next;
	if (select("Sair do salão:Ficar") == 2)
		close;
	close2;
	warp getvariableofnpc(.volta$, "CheffeniaPorta"),
	     getvariableofnpc(.volta_x, "CheffeniaPorta"),
	     getvariableofnpc(.volta_y, "CheffeniaPorta");
	end;

OnInit:
	waitingroom "Saída de Cheffenia", 0;
	end;
}

//======================================================
// O Relogio - expulsa quem esta com o passe vencido
//
// Varre o mapa a cada 30s em vez de dar addtimer no jogador
// na entrada. O motivo esta no cabecalho: timer de jogador
// morre em relog e em @reloadscript, e ai o passe vencido
// nunca expiraria. A varredura sobrevive aos dois.
//======================================================
-	script	CheffeniaRelogio	-1,{
OnInit:
	initnpctimer;
	end;

OnTimer30000:
	.@mapa$ = getvariableofnpc(.mapa$, "CheffeniaPorta");
	.@volta$ = getvariableofnpc(.volta$, "CheffeniaPorta");
	.@vx = getvariableofnpc(.volta_x, "CheffeniaPorta");
	.@vy = getvariableofnpc(.volta_y, "CheffeniaPorta");
	.@agora = gettimetick(2);

	// getmapunits devolve os block ids; para jogador o block id
	// E o account id, entao attachrid recebe ele direto.
	.@n = getmapunits(BL_PC, .@mapa$, .@rid);
	for (.@i = 0; .@i < .@n; .@i++) {
		if (!attachrid(.@rid[.@i]))
			continue;
		// GM entra para administrar, nao para farmar: nao expulsa.
		if (getgroupid() < 99 && cheffenia_ate <= .@agora) {
			dispbottom "Seu passe de Cheffenia venceu.";
			warp .@volta$, .@vx, .@vy;
		}
		detachrid;
	}
	setnpctimer 0;
	end;
}
