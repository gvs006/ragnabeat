//===== rAthena Script =======================================
//= Teleportadora
//===== Descrição: ===========================================
//= Menu único de teleporte: cidades, campos, calabouços,
//= castelos e feudos de clã, instâncias e áreas especiais.
//= Guarda o último destino e o oferece no topo do menu.
//=
//= Base: script Warper do rAthena, por [Euphy]. O corpo (lógica,
//= coordenadas, Restrict/Disp/Pick) está intacto - só o texto
//= visível foi traduzido, mais a ordem alfabética dos menus de
//= topo, refeita porque a ordem do inglês não serve em PT-BR.
//=
//===== DE ONDE VEM CADA NOME ================================
//= Os nomes de mapa saem do System/mapInfo_true.lub do cliente
//= (fonte LATAM/bRO) - o mesmo texto que o jogador vê no canto
//= da tela ao chegar. Onde o cliente não tem nome (mapas só de
//= renewal recente: jor_*, icecastle, thor_camp), a tradução é
//= literal do original.
//=
//= Por isso alguns nomes fogem do inglês: Morocc virou Morroc,
//= Splendide virou Esplendor, Niflheim virou Nifflheim, Manuk
//= virou Manuka, Byalan virou Túnel Submarino, Coal Mines virou
//= Mina Abandonada. Não desfaça isso: quem joga lá o
//= nome que o cliente mostra, e divergir aqui é que confunde.
//=
//= Cidade continua com o nome próprio: Geffen, e não a descrição
//= que o cliente usa como displayName dela (A Cidade da Magia).
//=
//===== ARMADILHAS DESTE SCRIPT ==============================
//= 1. ENCODING: cp1252, como todo .txt de NPC deste repo. Ver
//=    docs/traducao.md. Gravar em UTF-8 quebra todo acento.
//=
//= 2. DOIS-PONTOS É SEPARADOR. Tanto o menu quanto o Disp()
//=    montam a lista com dois-pontos. Nenhum texto de opção
//=    pode
//=    conter dois-pontos, ou a opção se parte em duas e todo
//=    índice depois dela anda uma casa.
//=
//= 3. Restrict("Pre-RE",n,...) usa o ÍNDICE da opção no
//=    submenu, não o texto. Mudar a CONTAGEM de itens de um
//=    Disp() desalinha o bloqueio e manda o jogador para o
//=    mapa errado. Traduzir o texto é seguro; acrescentar ou
//=    remover item, não.
//=
//= 4. SERVIDOR PRE-RENEWAL. Tudo sob Restrict("RE") responde
//=    "Este mapa não está habilitado no Pre-Renewal" - boa
//=    parte do menu (Eclage, Malangdo, Rockridge, Verus, as
//=    instâncias novas) existe mas nunca abre. Podar essa lista
//=    é tarefa à parte, não de tradução.
//=

// ------------------- Funções -------------------
// * Go("<mapa>",<x>,<y>);
//	~ Teleporta direto para um mapa.
//
// * Disp("<Opção do menu>",<primeira>,<última>);
// * Pick("<prefixo do mapa>"{,<deslocamento do índice>});
//	~ Menu e escolha de mapa autonumerados. Campos e
//	  calabouços são obrigados a usar Disp e Pick.
//
// * Disp("<Opção 1>:<Opção 2>:<etc.>");
// * Pick("","<mapa1>","<mapa2>","<etc.>");
//	~ Menu e mapas listados à mão.
//
// * Restrict("<RE | Pre-RE>"{,<números de opção>});
//	~ Libera o mapa só em Renewal ou só em Pre-Renewal.
//     Com números de opção, só esses mapas são bloqueados
//     (ou seja, não vale para o "Go").
//
// Mais:
//   ~ O array @c[] guarda todas as coordenadas (x,y).
//   ~ Use @c[2], EXCETO quando os mapas começam em 0:
//	  aí use @c[0] e deslocamento 1 no Pick().
// --------------------------------------------------

function Go {
	set lastwarp$, getarg(0);
	set lastwarpx, getarg(1,0);
	set lastwarpy, getarg(2,0);
	warp getarg(0),getarg(1,0),getarg(2,0);
	end;
}
function Disp {
	if (getargcount() < 3)
		set @menu$, getarg(0);
	else {
		set @menu$,"";
		for (set .@i,getarg(1); .@i<=getarg(2); set .@i,.@i+1)
			set @menu$, @menu$+getarg(0)+" "+.@i+":";
	}
	return;
}
function Pick {
	set .@warp_block,@warp_block;
	set @warp_block,0;
	if((@f && .OnlyFirstFld) || (@d && .OnlyFirstDun)){
		set .@select,1;
		if(.@warp_block){
			while(.@warp_block & (1<<.@select)){
				.@select += 1;
			}
		}
	}else{
		set .@select, select(@menu$);
	}
	if (getarg(0) == "") {
		set .@i, .@select;
		set .@map$, getarg(.@i);
	} else {
		set .@i, .@select-getarg(1,0);
		set .@map$, getarg(0)+((.@i<10)?"0":"")+.@i;
	}
	if (.@warp_block & (1<<.@select)) {
		message strcharinfo(0),"Este mapa não está habilitado no "+((checkre(0))?"":"Pre-")+"Renewal.";
		end;
	}
	set .@x, @c[.@i*2];
	set .@y, @c[.@i*2+1];
	deletearray @c[0],getarraysize(@c);
	@f = false; @d = false;
	Go(.@map$,.@x,.@y);
}
function Restrict {
	.@type$ = getarg(0);
	
	if (.@type$ == "Brasilis") {
		// If the official warper to Brasilis is enabled, meaning is either
		// Renewal mode or Pre-renewal Brasilis is enabled, we lift the restriction
		if (getnpcid(0, "Crewman_bra2"))
			return;
		// Otherwise we apply the renewal restriction
		.@type$ = "RE";
	}

	if ((.@type$ == "RE" && !checkre(0)) || (.@type$ == "Pre-RE" && checkre(0))) {
		if (getarg(1,0)) {
			set @warp_block,0;
			for (set .@i,1; .@i<getargcount(); set .@i,.@i+1)
				set @warp_block, @warp_block | (1<<getarg(.@i));
		} else {
			message strcharinfo(0),"Este mapa não está habilitado no "+((checkre(0))?"":"Pre-")+"Renewal.";
			end;
		}
	}
	return;
}
