prt_in,167,169,4	script	Mestre do Câmbio	111,{
	.@coin = 674; // Mithril Coin
	.@price = 100000; // 100k cada moeda
	
	mes "[Mestre do Câmbio]";
	mes "Aqui nós não aceitamos papel sujo do reino.";
	mes "Se quiser comprar algo, vai precisar de ^FF0000Mithril Coins^000000.";
	mes "Eu vendo cada uma por ^00AA00" + .@price + " Zeny^000000.";
	next;
	mes "[Mestre do Câmbio]";
	mes "Quantas você quer?";
	input .@amount;
	
	if (.@amount <= 0) {
		mes "[Mestre do Câmbio]";
		mes "Cai fora, não brinque com meu tempo.";
		close;
	}
	
	.@total = .@amount * .@price;
	
	if (Zeny < .@total) {
		mes "[Mestre do Câmbio]";
		mes "Você não tem Zeny suficiente para isso.";
		close;
	}
	
	Zeny -= .@total;
	getitem .@coin, .@amount;
	mes "[Mestre do Câmbio]";
	mes "Aqui estão suas " + .@amount + " moedas. Agora vá gastar.";
	close;
}

// Função de Loja Customizada
function	script	F_EliteShop	{
	.@slot$ = getarg(0);
	.@coin = 674;
	.@price = 100; // Preço padrão: 100 moedas
	
	mes "[Vendedor de Elite - " + .@slot$ + "]";
	mes "Escolha a carta MVP que deseja para seu slot de " + .@slot$ + ".";
	mes "Cada uma custa ^FF0000" + .@price + " Mithril Coins^000000.";
	next;
	
	// Construir menu dinâmico
	.@menu$ = "";
	for (.@i = 1; .@i < getargcount(); .@i++) {
		.@menu$ += getitemname(getarg(.@i)) + ":";
	}
	
	.@sel = select(.@menu$) + 1; // +1 porque pulei o arg 0 (slot$)
	.@itemID = getarg(.@sel - 1);
	
	mes "[Vendedor de Elite]";
	mes "Tem certeza que deseja comprar a carta ^0000FF" + getitemname(.@itemID) + "^000000?";
	next;
	if (select("Sim, comprar.:Não.") == 2) {
		close;
	}
	
	if (countitem(.@coin) < .@price) {
		mes "[Vendedor de Elite]";
		mes "Você não tem moedas suficientes! Vá falar com o Mestre do Câmbio.";
		close;
	}
	
	delitem .@coin, .@price;
	getitem .@itemID, 1;
	specialeffect2 EF_LEVELUP;
	mes "[Vendedor de Elite]";
	mes "Excelente escolha! Aqui está sua carta.";
	close;
}
