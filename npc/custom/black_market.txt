//===== rAthena Script =======================================
//= Mercado Negro - Contrabandista
//===== Descrição: ===========================================
//= Um único NPC em Prontera, com menu de dois níveis:
//=
//=   1) a parte do corpo  (Elmo, Arma, Armadura, ...)
//=   2) o tipo            (Cartas MvP, Cartas Normais, Equips,
//=                         Equips 3rd)
//=
//= Cada folha abre uma loja de troca nativa (barter), onde o
//= jogador vê o ícone e a descrição, escolhe a quantidade, e
//= o servidor debita a moeda com checagem de peso e espaço.
//=
//= AS LISTAS DE ITEM NÃO ESTÃO AQUI. Elas ficam em
//= npc/custom/barters.yml, no formato Contra#<tipo>_<slot>.
//= Este script só monta o nome e chama.
//=
//= O que "Equips 3rd" quer dizer aqui: equipamento nascido no
//= episódio das 3as classes e trazido para o pre-renewal, com
//= nível e classe ajustados. Não existe 3a classe neste
//= servidor. Os valores e o porquê de cada ajuste estão em
//= docs/rebalance.md.
//=
//= Por que barter e não itemshop: o itemshop do rAthena abre
//= a janela de Cash Shop (src/map/npc.cpp, npc_click ->
//= clif_cashshop_show). No cliente 2025 ela mostra campo de
//= "Free Cash" e erra a mensagem de saldo, falando em Kafra
//= Points quando na verdade faltam moedas. O barter é a
//= janela feita para moeda-item e mostra o que falta certo.
//=
//= Moeda: Mithril Coin (674). Peso zerado em
//= db/ragnabeat_items.yml para não pesar no inventário.
//============================================================

//= BONUS ALEATORIO: nada comprado aqui vem com BA. O barter
//= entrega o item cru - ele nao passa pelo randopt_drop_*
//= (que so vale para drop de monstro) nem pelo
//= randomoptgroup() de script. Para vender com BA seria
//= preciso trocar barter por shop + npcshopattach +
//= OnBuyItem, e ai some a janela com icone e quantidade.
//= Ver docs/rebalance.md.
//============================================================

prontera,143,171,4	script	Contrabandista	615,{
	mes "[Contrabandista]";
	mes "Psiu... aqui dentro não circula papel do reino.";
	mes "Só aceito " + mesitemlink(.Moeda) + ", e você tem ^0000FF" + countitem(.Moeda) + "^000000.";
	next;

L_Slot:
	.@menu$ = "";
	for (.@i = 0; .@i < getarraysize(.Slot$); .@i++)
		.@menu$ += .Slot$[.@i] + ":";
	.@menu$ += "Comprar " + getitemname(.Moeda) + ":Ir embora";

	.@s = select(.@menu$) - 1;
	if (.@s == getarraysize(.Slot$))
		goto L_Cambio;
	if (.@s < 0 || .@s > getarraysize(.Slot$)) {
		mes "[Contrabandista]";
		mes "Então some daqui.";
		close;
	}

	// .Tipos[] e uma mascara, ver o OnInit.
	// Categoria vazia nao vira opcao - nada de porta para sala vazia.
	.@m = .Tipos[.@s];
	.@n = 0;
	.@menu$ = "";
	deletearray .@pref$[0];
	for (.@i = 0; .@i < getarraysize(.Tipo$); .@i++) {
		if (!(.@m & (1 << .@i)))
			continue;
		.@menu$ += .Tipo$[.@i] + " [" + .Slot$[.@s] + "]:";
		.@pref$[.@n] = .TipoId$[.@i];
		.@n++;
	}
	.@menu$ += "Voltar";

	.@t = select(.@menu$) - 1;
	if (.@t < 0 || .@t >= .@n) {
		mes "[Contrabandista]";
		mes "Decida-se.";
		next;
		goto L_Slot;
	}

	close2;
	callshop "Contra#" + .@pref$[.@t] + "_" + .SlotId$[.@s];
	end;

L_Cambio:
	mes "[Contrabandista]";
	mes "Cada " + mesitemlink(.Moeda) + " sai por ^00AA00" + callfunc("F_InsertComma", .Preco) + " Zeny^000000.";
	mes "Quantas você quer? (máximo " + .MaxCompra + " por vez)";
	next;

	input .@qtd, 0, .MaxCompra;
	if (.@qtd < 1) {
		mes "[Contrabandista]";
		mes "Não brinque com o meu tempo.";
		close;
	}

	.@total = .@qtd * .Preco;
	if (Zeny < .@total) {
		mes "[Contrabandista]";
		mes "Isso daria ^00AA00" + callfunc("F_InsertComma", .@total) + " Zeny^000000.";
		mes "Você não tem.";
		close;
	}

	Zeny -= .@total;
	getitem .Moeda, .@qtd;
	mes "[Contrabandista]";
	mes "Aqui estão suas " + .@qtd + " moedas. Agora vê gastar.";
	close;

OnInit:
	waitingroom "Contrabandista", 0;
	.Moeda = 674;			// Mithril Coin
	.Preco = 100000;		// Zeny por moeda
	.MaxCompra = 500;		// teto por compra (500 x 100k = 50M, longe do estouro de Zeny)

	// Nome exibido e sufixo usado no nome da loja do barters.yml.
	setarray .Slot$[0],
		"Elmo", "Arma", "Armadura", "Escudo", "Capa", "Calçado", "Acessório";
	setarray .SlotId$[0],
		"elmo", "arma", "armadura", "escudo", "capa", "calcado", "acessorio";

	setarray .Tipo$[0],
		"Cartas MvP", "Cartas Normais", "Equips", "Equips 3rd";
	setarray .TipoId$[0],
		"mvp",        "norm",           "eq",     "3rd";

	// Que tipos existem em cada slot, na ordem de .Slot$.
	// 1 = Cartas MvP, 2 = Cartas Normais, 4 = Equips, 8 = Equips 3rd.
	// ISTO TEM QUE BATER COM npc/custom/barters.yml. Categoria marcada aqui
	// sem loja la vira "Shop not found" no console quando o jogador clicar.
	setarray .Tipos[0],
		1|2|4|8,	// Elmo
		1|2|4|8,	// Arma
		1|2|4,		// Armadura
		1|2,		// Escudo
		1|2|4|8,	// Capa
		1|4,		// Calcado   (sem carta normal)
		1|2|4|8;	// Acessorio
	end;
}
