//===== rAthena Script =======================================
//= @restock — Reabastecimento manual com janela
//===== Por: =================================================
//= Ideia original: Mastagoon (rAthena topic 127084)
//= Reescrito para Ragnabeat com interface de dialogo
//===== Descrição: ===========================================
//= O jogador monta uma lista de itens com quantidade-alvo.
//= Ao usar @restock, abre uma janela mostrando a lista atual
//= no topo e as opções logo abaixo. O reabastecimento puxa do
//= armazém apenas o que está faltando para atingir o alvo.
//=
//= Toda a configuração é pela janela — nada de decorar
//= sub-comando de chat.
//===== ATENÇÃO — VIP: =======================================
//= Esta feature é EXCLUSIVA DE VIP por decisão de projeto
//= (ver ROADMAP.md, seções 2.1 e 3.1).
//=
//= O gate é o F_EhVip de npc/custom/vip.txt — não duplique a
//= checagem aqui. É lá que se troca o modo quando a fase 3.1
//= entrar.
//============================================================

-	script	restock_atcmd	-1,{

// ---------------------------------------------------------
// Um item só pode entrar na lista se for de um tipo
// consumível. Equipamento e carta ficam de fora de propósito:
// puxar equipamento do armazém sem o jogador ver é receita
// para perda acidental.
// ---------------------------------------------------------
function	podeEntrar	{
	return (inarray(.AllowedTypes[0], getiteminfo(getarg(0), ITEMINFO_TYPE)) >= 0);
}

OnInit:
	// --- Item consumido a cada reabastecimento. 0 = de graça.
	.TicketID = 0;

	// --- Limites da lista
	.MaxQty = 500;
	.MaxItens = 10;

	// --- 1 = só funciona dentro de cidade
	.TownOnly = 0;

	// --- Tipos que podem entrar na lista
	setarray .AllowedTypes[0], IT_HEALING, IT_USABLE, IT_ETC, IT_AMMO, IT_DELAYCONSUME;

	bindatcmd "restock", strnpcinfo(3) + "::OnAtcommand";
	end;

OnAtcommand:
	if (!callfunc("F_EhVip")) {
		dispbottom "[Restock] Comando exclusivo para jogadores VIP.";
		end;
	}
	// Resquício da versão anterior, que tinha modo automático por timer.
	// Zerar aqui evita que a variável fique órfã no personagem para sempre.
	if (RestockAuto) RestockAuto = 0;

	// cai direto no menu

// =========================================================
// JANELA PRINCIPAL
// A lista atual fica no topo, as opções logo abaixo.
// =========================================================
L_Menu:
	clear;
	mes "^777777[ Reabastecimento ]^000000";
	mes " ";

	.@n = getarraysize(RestockList);
	if (!.@n) {
		mes "Sua lista está vazia.";
		mes "Use ^0055FFAdicionar item^000000 para montá-la.";
	} else {
		mes "Sua lista ^0055FF" + .@n + "^000000/" + .MaxItens + " — ^777777inventário / alvo^000000";
		for (.@i = 0; .@i < .@n; .@i++) {
			.@id = RestockList[.@i];
			.@alvo = RestockAmountList[.@i];
			.@tem = countitem(.@id);
			if (.@tem >= .@alvo)
				mes " " + mesitemlink(.@id) + " ^00AA00" + .@tem + "^000000 / " + .@alvo;
			else
				mes " " + mesitemlink(.@id) + " ^FF5555" + .@tem + "^000000 / " + .@alvo +
				    " ^777777(faltam " + (.@alvo - .@tem) + ")^000000";
		}
	}
	mes " ";

	switch (select(
		"Reabastecer agora",
		"Adicionar item",
		"Remover item",
		"Limpar a lista",
		"Fechar"
	)) {
		case 1: goto L_Reabastecer;
		case 2: goto L_Adicionar;
		case 3: goto L_Remover;
		case 4: goto L_Limpar;
		case 5: close;
	}
	end;

// =========================================================
// REABASTECER
// =========================================================
L_Reabastecer:
	clear;
	mes "^777777[ Reabastecimento ]^000000";
	mes " ";

	if (!getarraysize(RestockList)) {
		mes "Sua lista está vazia — não há o que puxar.";
		next;
		goto L_Menu;
	}
	if (.TownOnly && !getmapflag(strcharinfo(3), MF_TOWN)) {
		mes "Só é possível reabastecer dentro de cidades.";
		close;
	}
	if (.TicketID && !countitem(.TicketID)) {
		mes "Você precisa de 1x " + mesitemlink(.TicketID) + " para usar.";
		close;
	}

	.@puxou = 0;
	for (.@i = 0; .@i < getarraysize(RestockList); .@i++) {
		.@id = RestockList[.@i];
		.@falta = RestockAmountList[.@i] - countitem(.@id);
		if (.@falta <= 0)
			continue;

		// storagecountitem devolve -1 com o armazém aberto
		.@tem = storagecountitem(.@id);
		if (.@tem < 0) {
			mes "Feche o armazém antes de reabastecer.";
			close;
		}
		if (.@tem == 0) {
			mes " " + mesitemlink(.@id) + " ^FF5555nada no armazém^000000";
			continue;
		}
		if (.@tem < .@falta)
			.@falta = .@tem;

		// Sem esta checagem o getitem falha por peso, aborta o script e
		// emite ShowError no console do map-server.
		if (!checkweight(.@id, .@falta)) {
			mes " " + mesitemlink(.@id) + " ^FF5555peso insuficiente^000000";
			continue;
		}

		getitem .@id, .@falta;
		storagedelitem .@id, .@falta;
		.@puxou = 1;
		mes " " + mesitemlink(.@id) + " ^00AA00+" + .@falta + "^000000";
	}

	if (!.@puxou)
		mes "Nada a reabastecer — sua lista já está completa.";
	else if (.TicketID)
		delitem .TicketID, 1;

	next;
	goto L_Menu;

// =========================================================
// ADICIONAR
// =========================================================
L_Adicionar:
	clear;
	mes "^777777[ Adicionar item ]^000000";
	mes " ";

	if (getarraysize(RestockList) >= .MaxItens) {
		mes "Sua lista já tem o máximo de ^0055FF" + .MaxItens + "^000000 itens.";
		mes "Remova algum antes de adicionar outro.";
		next;
		goto L_Menu;
	}

	mes "De onde quer escolher o item?";
	mes " ";
	switch (select("Do meu inventário", "Digitar o ID do item", "Voltar")) {
		case 1: goto L_AddInventario;
		case 2: goto L_AddPorID;
		case 3: goto L_Menu;
	}
	end;

L_AddInventario:
	clear;
	mes "^777777[ Adicionar item ]^000000";
	mes " ";

	getinventorylist;
	.@n = 0;
	.@menu$ = "";
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		.@id = @inventorylist_id[.@i];
		if (!podeEntrar(.@id))
			continue;
		if (inarray(RestockList[0], .@id) >= 0)
			continue;
		// Item não-empilhável ocupa vários slots: sem esta checagem ele
		// apareceria repetido no menu.
		if (inarray(.@opt[0], .@id) >= 0)
			continue;
		.@menu$ = .@menu$ + (.@n ? ":" : "") + getitemname(.@id);
		.@opt[.@n] = .@id;
		.@n++;
	}

	if (!.@n) {
		mes "Não há nada no seu inventário que possa entrar na lista.";
		mes " ";
		mes "^777777Só entram consumíveis: itens de cura, usáveis,";
		mes "matéria-prima e munição.^000000";
		next;
		goto L_Menu;
	}

	mes "Escolha o item:";
	.@sel = select(.@menu$ + ":^777777Voltar^000000");
	if (.@sel > .@n)
		goto L_Menu;

	.@novo = .@opt[.@sel - 1];
	goto L_AddQuantidade;

L_AddPorID:
	clear;
	mes "^777777[ Adicionar item ]^000000";
	mes " ";
	mes "Digite o ID do item.";
	mes "^777777Útil para item que está só no armazém,";
	mes "que o servidor não consegue listar aqui.^000000";
	next;

	input .@novo, 0, 65535;

	clear;
	mes "^777777[ Adicionar item ]^000000";
	mes " ";
	if (getitemname(.@novo) == "null") {
		mes "Não existe item com o ID ^FF5555" + .@novo + "^000000.";
		next;
		goto L_Menu;
	}
	if (!podeEntrar(.@novo)) {
		mes mesitemlink(.@novo) + " não pode ser reabastecido.";
		mes " ";
		mes "^777777Só entram consumíveis: itens de cura, usáveis,";
		mes "matéria-prima e munição.^000000";
		next;
		goto L_Menu;
	}
	if (inarray(RestockList[0], .@novo) >= 0) {
		mes mesitemlink(.@novo) + " já está na sua lista.";
		next;
		goto L_Menu;
	}
	// cai para a quantidade

L_AddQuantidade:
	clear;
	mes "^777777[ Adicionar item ]^000000";
	mes " ";
	mes "Item: " + mesitemlink(.@novo);
	mes " ";
	mes "Quantos você quer manter sempre no inventário?";
	mes "^777777Entre 1 e " + .MaxQty + ".^000000";
	next;

	// input com min/max já limita o valor, não precisa revalidar
	input .@qtd, 1, .MaxQty;

	setarray RestockList[getarraysize(RestockList)], .@novo;
	setarray RestockAmountList[getarraysize(RestockAmountList)], .@qtd;

	clear;
	mes "^777777[ Adicionar item ]^000000";
	mes " ";
	mes mesitemlink(.@novo) + " x^0055FF" + .@qtd + "^000000 adicionado à lista.";
	next;
	goto L_Menu;

// =========================================================
// REMOVER
// =========================================================
L_Remover:
	clear;
	mes "^777777[ Remover item ]^000000";
	mes " ";

	.@n = getarraysize(RestockList);
	if (!.@n) {
		mes "Sua lista já está vazia.";
		next;
		goto L_Menu;
	}

	.@menu$ = "";
	for (.@i = 0; .@i < .@n; .@i++)
		.@menu$ = .@menu$ + (.@i ? ":" : "") + getitemname(RestockList[.@i]) +
		          " x" + RestockAmountList[.@i];

	mes "Qual item quer tirar da lista?";
	.@sel = select(.@menu$ + ":^777777Voltar^000000");
	if (.@sel > .@n)
		goto L_Menu;

	.@id = RestockList[.@sel - 1];
	deletearray RestockList[.@sel - 1], 1;
	deletearray RestockAmountList[.@sel - 1], 1;

	clear;
	mes "^777777[ Remover item ]^000000";
	mes " ";
	mes mesitemlink(.@id) + " removido da lista.";
	next;
	goto L_Menu;

// =========================================================
// LIMPAR
// =========================================================
L_Limpar:
	clear;
	mes "^777777[ Limpar a lista ]^000000";
	mes " ";

	if (!getarraysize(RestockList)) {
		mes "Sua lista já está vazia.";
		next;
		goto L_Menu;
	}

	mes "Apagar os ^0055FF" + getarraysize(RestockList) + "^000000 itens da lista?";
	mes "^777777Isso não mexe nos seus itens, só na configuração.^000000";
	if (select("Não, voltar", "Sim, apagar tudo") == 1)
		goto L_Menu;

	deletearray RestockList[0], getarraysize(RestockList);
	deletearray RestockAmountList[0], getarraysize(RestockAmountList);

	clear;
	mes "^777777[ Limpar a lista ]^000000";
	mes " ";
	mes "Lista apagada.";
	next;
	goto L_Menu;
}
